// Sistema multi-database esteso a Food, Beauty e General Products (No-Food) async function cercaSuOpenFoodFactsMultipli() { let barcode = document.getElementById('codiceBarre').value.trim(); if (!barcode) return; let msgInfo = document.getElementById('msgInfo'); msgInfo.textContent = "Interrogazione database multipli (Food, Beauty, No-Food) in corso..."; msgInfo.style.display = 'block'; let endpoints = [ `https://world.openfoodfacts.org/api/v0/product/${barcode}.json`, `https://world.openbeautyfacts.org/api/v0/product/${barcode}.json`, `https://world.openproducts.org/api/v0/product/${barcode}.json` ]; let trovato = false; for (let url of endpoints) { try { let response = await fetch(url); let data = await response.json(); if (data.status === 1 && data.product) { trovato = true; msgInfo.style.display = 'none'; let nome = data.product.product_name_it || data.product.product_name || ""; let marca = data.product.brands || ""; document.getElementById('nomeProdotto').value = (marca && nome) ? `${marca} - ${nome}` : (nome || marca); // Gestione intelligente scadenza (i non alimentari o prodotti senza scadenza alimentare impostano di default "Consigliata" o "Senza") let categorie = (data.product.categories_tags || []).join(" ").toLowerCase(); if (url.includes("openbeautyfacts") || url.includes("openproducts")) { // Prodotti no-food / cosmetici spesso hanno PAO (period after opening) o scadenza non tassativa alimentare impostaTipoScadenza('consigliata'); } else if (categorie.includes("fresh") || categorie.includes("fresco") || categorie.includes("milk") || categorie.includes("latte") || categorie.includes("meat") || categorie.includes("carne") || categorie.includes("fish") || categorie.includes("pesce")) { impostaTipoScadenza('tassativa'); } else { impostaTipoScadenza('consigliata'); } urlImmagineCorrente = data.product.image_front_small_url || data.product.image_front_url || ""; if (urlImmagineCorrente) { document.getElementById('imgProdotto').src = urlImmagineCorrente; document.getElementById('boxAnteprima').style.display = 'block'; } break; } } catch (e) {} } if (!trovato) { msgInfo.style.display = 'none'; alert("Prodotto non trovato nei database (Food, Beauty e No-Food). Usa il pulsante 'Prodotto senza codice' o inserisci i dati manualmente."); } }
